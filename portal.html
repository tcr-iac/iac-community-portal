function showFileOverlay(title, fileName) { const overlay = document.getElementById('table-overlay'); document.getElementById('table-overlay-title').textContent = title; document.querySelector('#table-overlay-box table').style.display = 'none'; let msgBox = document.getElementById('file-coming-soon'); if (!msgBox) { msgBox = document.createElement('div'); msgBox.id = 'file-coming-soon'; msgBox.style.cssText = 'padding:10px;text-align:center;color:#888;'; document.getElementById('table-overlay-box').appendChild(msgBox); } const filePath = `files/${fileName}`; const ext = fileName.split('.').pop().toLowerCase(); fetch(filePath, { method: 'HEAD' }) .then(res => { if (res.ok) { if (ext === 'pdf') { window.open(filePath, '_blank'); return; } else if (['jpg','jpeg','png','gif','webp'].includes(ext)) { msgBox.innerHTML = ``; } else if (['ppt','pptx','doc','docx','xls','xlsx'].includes(ext)) { msgBox.innerHTML = `
📄

${fileName}

Download File
`; } else { msgBox.innerHTML = `Open ${fileName}`; } msgBox.style.display = 'block'; overlay.hidden = false; } else { showComingSoon(); } }) .catch(() => showComingSoon()); function showComingSoon() { msgBox.innerHTML = `
📄

${fileName}

This file is not loaded yet — coming soon!

`; msgBox.style.display = 'block'; overlay.hidden = false; } }
